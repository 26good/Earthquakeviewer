・音を下のscriptに変更してください。

<script>
    let audioCtx = null;
    let activeNodes = [];

    function initAudio() {
        if (!audioCtx) {
            audioCtx = new (window.AudioContext || window.webkitAudioContext)();
        }
    }

    function updateStatus(text, color, bgColor = "#1a1a1a") {
        const statusText = document.getElementById('statusText');
        statusText.innerText = text;
        statusText.style.color = color;
        statusText.style.backgroundColor = bgColor;
    }

    function stopAllSounds() {
        activeNodes.forEach(node => {
            try { node.stop(); } catch(e) {}
        });
        activeNodes = [];
    }

    // --- イベントリスナー ---
    document.getElementById('eewForecast').addEventListener('click', () => {
        initAudio();
        stopAllSounds();
        updateStatus("🔵 EEW第1報（予報）を受信", "#3498db");
        playEEWForecast();
    });

    document.getElementById('eewUpdate').addEventListener('click', () => {
        initAudio();
        updateStatus("🔄 EEW情報更新を受信", "#3498db");
        playEEWUpdate();
    });

    document.getElementById('eewFinal').addEventListener('click', () => {
        initAudio();
        stopAllSounds();
        updateStatus("✅ EEW最終報を受信", "#2ecc71");
        playEEWFinal();
    });

    document.getElementById('eewWarning').addEventListener('click', () => {
        initAudio();
        stopAllSounds();
        updateStatus("⚠️ 警報発信中（強い揺れ・津波に最大警戒）", "#fff", "#c0392b");
        playEEWWarning();
    });


    // --- 音源生成メインロジック ---

    // 1. EEW予報【修正版：1マス = 0.10秒、比率2:1:1:2でジャスト固定】
    function playEEWForecast() {
        const now = audioCtx.currentTime;
        
        // ★ ご指定通り1マスの基準時間を 0.1秒 に設定
        const unit = 0.10; 
        
        // 比率 (2 : 1 : 1 : 2)
        const soDuration = unit * 2; // ソ：2マス分 (0.20s)
        const doDuration = unit * 1; // ド：1マス分 (0.10s)
        const gap        = unit * 1; // 隙間：1マス分 (0.10s)
        const miDuration = unit * 2; // ミ：2マス分 (0.20s)
        
        // 発音タイミング
        const soStart = now;
        const doStart = soStart + soDuration; // ソの終了と同時にドが発音
        const miStart = doStart + doDuration + gap; // ド終了後、0.1秒の隙間（無音）を挟んでミ

        createTone(784, soStart, soDuration, 'sine', 0.25); // ソ
        createTone(523, doStart, doDuration, 'sine', 0.25); // ド
        createTone(659, miStart, miDuration, 'sine', 0.25); // ミ
        
        const totalDuration = soDuration + doDuration + gap + miDuration;
        setTimeout(() => updateStatus("ステーション：待機中", "#888"), (totalDuration + 0.2) * 1000);
    }

    // 2. EEW更新報（ミ単音）
    function playEEWUpdate() {
        const now = audioCtx.currentTime;
        createTone(659, now, 0.15, 'sine', 0.25);
    }

    // 3. EEW最終報（ミ・ド下降）
    function playEEWFinal() {
        const now = audioCtx.currentTime;
        createTone(659, now,       0.5, 'sine', 0.25);
        createTone(523, now + 0.2, 0.7, 'sine', 0.25);
        setTimeout(() => updateStatus("ステーション：待機中", "#888"), 1500);
    }

    // 4. EEW警報 / 津波警報（各和音0.50秒・トライアングル波）
    function playEEWWarning() {
        const now = audioCtx.currentTime;
        const volume = 0.18; 
        
        const duration = 0.50;
        const nextStart = 0.50; 
        const loopInterval = 1.0; 

        for (let i = 0; i < 3; i++) {
            const start = now + (i * loopInterval);

            // フレーズ前半（高音の和音）
            createTone(440, start, duration, 'triangle', volume);
            createTone(442.5, start, duration, 'triangle', volume);
            createTone(554, start, duration, 'triangle', volume);
            createTone(556, start, duration, 'triangle', volume);
            createTone(880, start, duration, 'triangle', volume);
            createTone(1108, start, duration, 'triangle', volume);

            // フレーズ後半（低音の和音)
            const start2 = start + nextStart;
            createTone(330, start2, duration, 'triangle', volume);
            createTone(331.5, start2, duration, 'triangle', volume);
            createTone(415, start2, duration, 'triangle', volume);
            createTone(417, start2, duration, 'triangle', volume);
            createTone(660, start2, duration, 'triangle', volume);
            createTone(830, start2, duration, 'triangle', volume);
        }
        setTimeout(() => updateStatus("ステーション：待機中", "#888", "#1a1a1a"), 3800);
    }


    // --- 統合・トーン生成関数 ---
    function createTone(frequency, startTime, duration, type, maxVolume = 0.25) {
        const oscillator = audioCtx.createOscillator();
        const gainNode = audioCtx.createGain();
        oscillator.type = type; 
        oscillator.frequency.setValueAtTime(frequency, startTime);
        
        // 音がさらに短くなったため、ブツブツ音（クリックノイズ）を防ぐ最適なフェード時間
        const fadeTime = Math.min(0.01, duration * 0.1); 
        
        gainNode.gain.setValueAtTime(0, startTime);
        gainNode.gain.linearRampToValueAtTime(maxVolume, startTime + fadeTime);
        gainNode.gain.setValueAtTime(maxVolume, startTime + duration - fadeTime);
        gainNode.gain.linearRampToValueAtTime(0, startTime + duration);
        
        oscillator.connect(gainNode);
        gainNode.connect(audioCtx.destination);
        oscillator.start(startTime);
        oscillator.stop(startTime + duration);
        activeNodes.push(oscillator);
    }
</script>
